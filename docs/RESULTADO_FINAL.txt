🚀 EXEMPLO DE USO - MÍNIMO

from brainbridge_v2.communication import UnityCommunicator, PatientData, TaskType

# Criar comunicador
comm = UnityCommunicator()
comm.start_server()

# Iniciar sessão
patient = PatientData("João Silva", 5, "Direito")
comm.start_session(patient, TaskType.TREINO)

# Enviar trigger para iniciar
comm.send_trigger()

# Executar ações
comm.send_hand_close("direita")
comm.send_flower_action("esquerda")

# Finalizar
comm.end_task("Treino concluído!")

# Parar servidor
comm.stop_server()
